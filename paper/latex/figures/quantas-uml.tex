% QUANTAS architecture, extended with Concrete QUANTAS.
% The original boxes, names and arrows are those of Figure 1 of the QUANTAS paper (ApPLIED '22);
% class names in parentheses, blue = Abstract only, orange = added by Concrete QUANTAS, white = shared.
% About 18.5 x 20 cm: fits one A4 page (scale it to \textwidth if needed).
% Needs \usepackage{tikz} and \usetikzlibrary{arrows.meta} in the preamble.
\begin{tikzpicture}[
    x=1cm, y=1cm, font=\sffamily\scriptsize, >={Straight Barb[length=1.8mm]},
    box/.style={draw, thick, align=center, text width=#1, inner sep=3pt, minimum height=1.1cm},
    box/.default=3.0cm,
    prog/.style={box=#1, rounded corners=4pt},
    prog/.default=3.0cm,
    abs/.style={fill=blue!8},
    con/.style={fill=orange!15},
    inherit/.style={thick, -{Triangle[open, length=2.8mm, width=2.8mm]}},
    realize/.style={inherit, dashed},
    compose/.style={thick, {Diamond[fill=black, length=3mm, width=2mm]}->},
    aggregate/.style={thick, {Diamond[open, length=3mm, width=2mm]}->},
    assoc/.style={thick, ->},
    depend/.style={thick, dashed, ->},
    line/.style={thick},
    dline/.style={thick, dashed},
    tag/.style={midway, above, inner sep=2pt, font=\sffamily\tiny},
  ]
  \newcommand{\comp}[2]{{\small #1}\\[-1pt]{\tiny #2}}
  \newcommand{\umltag}[1]{\guillemotleft #1\guillemotright}

  % ---------------- Original QUANTAS components (Figure 1 of the QUANTAS paper) ----------------
  \node[box=3.4cm, abs] (simulation) at (2.0, 20.0)
    {\comp{Simulation}{configures simulation, sets up logs, initializes network topology and state, runs simulation}};
  \node[box=3.4cm, abs] (network) at (2.0, 17.8)
    {\comp{Network}{determines network topology: connects nodes and channels, sets up channel delays}};
  \node[box=3.4cm] (node) at (2.0, 15.6)
    {\comp{Abstract Node}{(\texttt{Peer}) node computation (command) interface}};
  \node[box=3.4cm] (nni) at (2.0, 13.2)
    {\comp{Node Network Interface}{(\texttt{NetworkInterface}) handles sending and receiving of packets (messages) between nodes}};
  \node[box=3.4cm] (packet) at (2.0, 10.8)
    {\comp{Packet}{packet header: source, destination, packet delay}};

  % User-provided components.
  \node[box] (config) at (7.6, 17.8)
    {\comp{Configuration}{processing config files: topology, delay, size, simulation length, etc.}};
  \node[box] (cnode) at (7.6, 14.4)
    {\comp{Concrete Node}{(\texttt{CPAPeer}, \texttt{DolevPeer}) algorithm-specific commands and variables, libraries}};
  \node[box] (message) at (7.6, 10.8)
    {\comp{User-Defined Message}{message payload}};
  \node[box=2.5cm, con] (adapter) at (11.7, 14.4)
    {\comp{Algorithm Adapter}{(\texttt{CPAPeerAdapter}) finished? values to report}};
  \draw[thick, dashed, rounded corners=10pt] (5.4, 9.8) rectangle (13.4, 19.0);
  \node[anchor=north east] at (13.3, 18.95) {\umltag{user-provided}};

  % ---------------- Added by Concrete QUANTAS ----------------
  \node[prog, con] (leader) at (16.2, 20.0)
    {\comp{Leader Process}{(\texttt{LeaderMain}) checks the input, plans the topology, writes the report}};
  \node[prog, con] (peerproc) at (16.2, 16.8)
    {\comp{Peer Process}{(\texttt{PeerMain}, \texttt{RunLoop}) one per node: receive, compute, send}};
  \node[box=2.6cm, con] (backend) at (10.4, 7.6)
    {\comp{Backend}{builds the coordinator and the node network interface}};
  \node[box=2.6cm, con] (coordinator) at (15.6, 7.6)
    {\comp{Coordinator}{control messages between leader and peers}};

  % Implementations.
  \node[box=2.1cm, abs] (nniabs) at (1.3, 4.4)
    {\comp{Abstract}{channel head/tail, implementing message delay}};
  \node[box=2.1cm, con] (nnibmq) at (3.9, 4.4)
    {\comp{BoostMq}{one shared-memory queue per node}};
  \node[box=2.1cm, con] (nnizmq) at (6.5, 4.4)
    {\comp{ZeroMQ}{PULL inbox, PUSH to each neighbour}};
  \node[box=2.1cm, con] (backbmq) at (9.1, 4.4) {\comp{BoostMq}{Boost.Interprocess}};
  \node[box=2.1cm, con] (backzmq) at (11.7, 4.4) {\comp{ZeroMQ}{\texttt{ipc://} or \texttt{tcp://}}};
  \node[box=2.1cm, con] (coordbmq) at (14.3, 4.4) {\comp{BoostMq}{control queues}};
  \node[box=2.1cm, con] (coordzmq) at (16.9, 4.4) {\comp{ZeroMQ}{ROUTER and DEALER}};
  \node[box=3.0cm, con] (measured) at (5.2, 1.4)
    {\comp{Measured Transport}{counts sent, received, dropped}};

  % Legend.
  \node[draw, abs, minimum size=2.5mm, inner sep=0pt] (la) at (10.4, 1.9) {};
  \node[anchor=west] at (la.east) {Abstract QUANTAS only};
  \node[draw, con, minimum size=2.5mm, inner sep=0pt] (lc) at (10.4, 1.4) {};
  \node[anchor=west] at (lc.east) {added by Concrete QUANTAS};
  \node[draw, minimum size=2.5mm, inner sep=0pt] (ls) at (10.4, 0.9) {};
  \node[anchor=west] at (ls.east) {shared by both};

  % ---------------- Relations of the original figure ----------------
  \draw[compose] (simulation) -- (network);
  \draw[compose] (network) -- (node);
  \draw[aggregate] (node) -- (nni);
  \draw[assoc] (nni) -- (packet);
  \draw[assoc] (simulation.east) -- (4.4, 20.0) |- (config.west);
  \draw[inherit] (cnode.west) -| ([xshift=1.0cm]node.south);
  \draw[compose] (cnode) -- (message);
  \draw[compose] (packet) -- (message);

  % ---------------- Relations added by Concrete QUANTAS ----------------
  \draw[assoc] (leader.west) -| (config.north);
  \draw[compose] (peerproc.west) -- (4.8, 16.8) |- (node.east);
  \draw[depend] ([yshift=-0.3cm]peerproc.west) -- (13.9, 16.5) |- (adapter.east);
  \draw[depend] (adapter.west) -- node[tag] {\umltag{reads}} (cnode.east);
  % Both processes use the Backend and the Coordinator (shared bus at y = 9.0).
  \draw[dline] (leader.east) -- (18.4, 20.0) -- (18.4, 9.0) -- (10.4, 9.0);
  \draw[dline] (peerproc.south) -- (16.2, 9.0);
  \fill (16.2, 9.0) circle (0.6mm);
  \fill (15.6, 9.0) circle (0.6mm);
  \draw[depend] (10.4, 9.0) -- (backend.north);
  \draw[depend] (15.6, 9.0) -- (coordinator.north);
  \draw[depend] (backend.east) -- node[tag] {\umltag{create}} (coordinator.west);

  % Inheritance of the implementations.
  \foreach \child in {nniabs, nnibmq, nnizmq}
    \draw[line] (\child.north) -- (\child.north |- 0, 6.0);
  \draw[inherit] (6.5, 6.0) -- (0.0, 6.0) |- (nni.west);
  \foreach \child in {backbmq, backzmq}
    \draw[inherit] (\child.north) -- (\child.north |- 0, 6.0) -| (backend.south);
  \foreach \child in {coordbmq, coordzmq}
    \draw[inherit] (\child.north) -- (\child.north |- 0, 6.0) -| (coordinator.south);
  \foreach \child in {nnibmq, nnizmq}
    \draw[realize] (\child.south) -- (\child.south |- 0, 2.7) -| (measured.north);
\end{tikzpicture}
