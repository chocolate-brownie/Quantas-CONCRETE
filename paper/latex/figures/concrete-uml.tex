% UML class diagram of Concrete QUANTAS, in the style of Figure 1 of the QUANTAS paper.
% Full text width: use inside figure*, with \input{figures/concrete-uml}.
% Needs \usepackage{tikz} and \usetikzlibrary{arrows.meta} in the preamble.
\begin{tikzpicture}[
    x=1cm, y=1cm, font=\scriptsize,
    cls/.style={draw, align=flush center, text width=#1, inner sep=2pt, minimum height=1.1cm},
    prog/.style={cls=#1, rounded corners=4pt, fill=black!6},
    inherit/.style={-{Triangle[open, length=2.4mm, width=2.4mm]}},
    compose/.style={{Diamond[fill=black, length=2.8mm, width=1.9mm]}-},
    depend/.style={dashed, -{Straight Barb[length=1.6mm]}},
    tag/.style={midway, above, inner sep=1.5pt},
  ]
  \newcommand{\umlname}[1]{\textbf{#1}}
  \newcommand{\umltag}[1]{\guillemotleft #1\guillemotright}

  % Programs (not classes).
  \node[prog=3.2cm] (peermain) at (5.69, 10.0) {\umlname{PeerMain + RunLoop}\\Peer process: receive, compute, repeat};
  \node[prog=3.2cm] (leadermain) at (5.69, 8.0) {\umlname{LeaderMain}\\Leader process: plans the topology, writes the report};

  % Interfaces and base classes.
  \node[cls=2.75cm] (coordinator) at (2.93, 5.8) {\umltag{interface}\\\umlname{Coordinator}\\Leader and peer control messages};
  \node[cls=2.75cm] (backend) at (8.445, 5.8) {\umltag{interface}\\\umlname{Backend}\\Builds the coordinator and the transport};
  \node[cls=2.75cm] (peer) at (12.605, 8.0) {\umltag{abstract}\\\umlname{Peer}\\Runs the algorithm};
  \node[cls=2.75cm] (netif) at (12.605, 5.8) {\umltag{abstract}\\\umlname{NetworkInterface}\\Sends and receives packets};
  \node[cls=2.1cm] (packet) at (16.2, 5.8) {\umlname{Packet}\\Source, target, message};
  \node[cls=3.2cm] (measured) at (14.225, 1.0) {\umltag{interface}\\\umlname{MeasuredTransport}\\Reports sent, received, dropped};

  % Backend implementations.
  \node[cls=2.75cm] (coordbmq) at (1.445, 3.3) {\umlname{BoostMqCoordinator}\\Control queues};
  \node[cls=2.4cm] (coordzmq) at (4.41, 3.3) {\umlname{ZMqCoordinator}\\ROUTER and DEALER};
  \node[cls=2.3cm] (backbmq) at (7.15, 3.3) {\umlname{BoostMqBackend}\\Boost.Interprocess queues};
  \node[cls=2.1cm] (backzmq) at (9.74, 3.3) {\umlname{ZMqBackend}\\\texttt{ipc://} or \texttt{tcp://}};
  \node[cls=2.85cm] (netbmq) at (12.605, 3.3) {\umlname{NetworkInterfaceBMq}\\One data queue per peer};
  \node[cls=2.85cm] (netzmq) at (15.845, 3.3) {\umlname{NetworkInterfaceZMq}\\PULL inbox, one PUSH per neighbour};

  % User-provided algorithms.
  \node[cls=1.9cm, minimum height=0.9cm] (cpa) at (16.1, 9.2) {\umlname{CPAPeer}};
  \node[cls=1.9cm, minimum height=0.9cm] (dolev) at (16.1, 8.0) {\umlname{DolevPeer}};
  \draw[dashed] (14.9, 7.25) rectangle (17.34, 10.55);
  \node[anchor=north] at (16.12, 10.55) {\umltag{user-provided}};

  % Inheritance.
  \foreach \child/\parent in {coordbmq/coordinator, coordzmq/coordinator, backbmq/backend, backzmq/backend, netbmq/netif, netzmq/netif}
    \draw[inherit] (\child.north) -- (\child.north |- 0,4.5) -- (\parent.south |- 0,4.5) -- (\parent.south);
  \foreach \child in {netbmq, netzmq}
    \draw[inherit] (\child.south) -- (\child.south |- 0,2.2) -- (measured.north |- 0,2.2) -- (measured.north);
  \foreach \child in {cpa, dolev}
    \draw[inherit] (\child.west) -| (14.5, 8.0) -- (peer.east);

  % Composition.
  \draw[compose] (peer.south) -- (netif.north);
  \draw[compose] (netif.east) -- (packet.west);

  % Creation.
  \draw[depend] (backend.west) -- node[tag] {\umltag{create}} (coordinator.east);
  \draw[depend] (backend.east) -- node[tag] {\umltag{create}} (netif.west);

  % Dependencies of the programs.
  \draw[depend] (leadermain.west) -| ([xshift=5mm]coordinator.north);
  \draw[depend] (leadermain.east) -| ([xshift=-5mm]backend.north);
  \draw[depend] (peermain.west) -| ([xshift=-5mm]coordinator.north);
  \draw[depend] ([yshift=-2mm]peermain.east) -| ([xshift=5mm]backend.north);
  \draw[depend] ([yshift=2mm]peermain.east) -| (peer.north);
\end{tikzpicture}
